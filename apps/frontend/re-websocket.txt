    DetectionEvent
}

type MessageCallback =
    (
        event:DetectionEvent
    )=>void




export class WebSocketService {


    private socket:
        WebSocket | null = null



    private url:string



    private callback:
        MessageCallback | null = null



    constructor(
        url:string
    ){

        this.url=url

    }



    connect(){


        if(this.socket){

            return

        }



        this.socket =
            new WebSocket(
                this.url
            )

            console.log(
                "WebSocket connected:",
                this.url
            )

        }

        (
            message
        )=>{

                    JSON.parse(
                        message.data
                    )

                        data
                    )

                }

                console.error(
                    "WebSocket parse error",
                    error
                )
